cmake_minimum_required(VERSION 3.24)
project(muscriptor_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MUSCRIPTOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_BUILD_BENCH OFF CACHE BOOL "" FORCE)
option(ANDROID_MUSCRIPTOR_VULKAN "Enable Vulkan" ON)
set(MUSCRIPTOR_VULKAN ${ANDROID_MUSCRIPTOR_VULKAN} CACHE BOOL "" FORCE)
add_subdirectory(vendor/muscriptor/cpp ${CMAKE_BINARY_DIR}/muscriptor)
add_library(muscriptor_jni SHARED bridge.cpp)
target_link_libraries(muscriptor_jni PRIVATE muscriptor_ggml android log)
set_target_properties(muscriptor_jni PROPERTIES OUTPUT_NAME "muscriptor_jni")
